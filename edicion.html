function render(list) { visibleProjects = list; if (!list.length) { GRID.innerHTML = `

No se encontraron proyectos.

`; return; } GRID.innerHTML = list.map((p, i) => { const title = escapeHTML( p.title || "Proyecto" ); const desc = escapeHTML( p.desc || "" ); const category = escapeHTML( p.categoria || "Edición" ); /* ======================================== PORTADA DE LA TARJETA ======================================== */ let mediaCard = ""; /* Si el proyecto tiene una miniatura definida en "cover", se usa normalmente. */ if (p.cover) { mediaCard = ` `; } /* Si NO existe cover, pero existe un MP4, usamos el propio video como miniatura. NO se reproduce. NO tiene controles. NO tiene sonido. */ else if ( p.video && /\.mp4$/i.test(p.video) ) { mediaCard = ` `; } /* Si no tiene imagen ni MP4, mostramos el placeholder. */ else { mediaCard = ` `; } return `
${mediaCard}
${category}

${title}

${desc}

Ver proyecto
`; }).join(""); /* ======================================== SI UNA IMAGEN FALLA ======================================== */ GRID .querySelectorAll("img.poster") .forEach(img => { img.addEventListener( "error", () => { if (img.dataset.fallback) { img.src = img.dataset.fallback; } }, { once: true } ); }); /* ======================================== CONVERTIR VIDEO EN "MINIATURA" ======================================== */ GRID .querySelectorAll(".static-video") .forEach(video => { /* Una vez que conocemos la duración del video, vamos al segundo 3. */ video.addEventListener( "loadedmetadata", () => { try { /* Si el video dura más de 3 segundos, usamos el segundo 3. Si dura menos, usamos aproximadamente la mitad. */ if (video.duration > 3) { video.currentTime = 3; } else if (video.duration > 0) { video.currentTime = video.duration / 2; } } catch (error) { console.log( "No se pudo cargar el fotograma:", error ); } } ); /* Nos aseguramos de que nunca empiece a reproducirse. */ video.pause(); }); }
